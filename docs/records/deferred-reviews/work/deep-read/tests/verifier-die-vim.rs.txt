//! Verification of the deep-read findings (scratch, not to land as is).

use die_core::{Ctx, State};
use die_vim::Vim;

fn run(text: &str, cursor: usize, keys: &str) -> (String, usize, State, Vec<String>) {
    let mut ctx = Ctx::new(vec![Box::new(Vim::default())]).expect("die-vim registers");
    let b = ctx.create_text("t", text);
    let w = ctx.open_window(b, 80, 24).expect("a window");
    ctx.focus(w);
    assert_eq!(ctx.text_mut(b).unwrap().set_cursor(w, cursor), Some(cursor));
    ctx.type_keys(keys);
    let t = ctx.text(b).unwrap();
    (
        t.rope().to_string(),
        t.cursor(w).unwrap(),
        ctx.state(),
        ctx.messages.clone(),
    )
}
fn cursor_after(text: &str, cursor: usize, keys: &str) -> usize {
    run(text, cursor, keys).1
}

#[test]
fn dr01_u_returns_to_where_the_steps_first_change_was_made() {
    assert_eq!(cursor_after("abc\n", 0, "ox<Esc>u"), 0, "ox<Esc>u from col 0");
    assert_eq!(cursor_after("abc\ndef\n", 1, "ox<Esc>u"), 1, "ox<Esc>u from col 1");
    assert_eq!(cursor_after("abc\ndef\n", 5, "Ox<Esc>u"), 5, "Ox<Esc>u from col 1");
    assert_eq!(cursor_after("abc\n", 1, "3ox<Esc>u"), 1, "3ox<Esc>u");
    assert_eq!(
        cursor_after("ab\ncd\n", 0, "jAx<Up>y<Esc>u"),
        4,
        "a session's later change to the left"
    );
}

#[test]
fn dr01_each_case() {
    let cases: &[(&str, usize, &str, usize)] = &[
        ("abc\n", 0, "ox<Esc>u", 0),
        ("abc\ndef\n", 1, "ox<Esc>u", 1),
        ("abc\ndef\n", 5, "Ox<Esc>u", 5),
        ("abc\n", 1, "3ox<Esc>u", 1),
        ("ab\ncd\n", 0, "jAx<Up>y<Esc>u", 4),
        ("abcd\nef\n", 0, "jAx<Up>y<Esc>u", 6),
        ("abc\n", 0, "i<Right>x<Esc>u", 1),
        ("abc\n", 1, "Ax<Esc>u", 2),
        ("  abc\n", 3, "Ix<Esc>u", 2),
    ];
    let mut bad = vec![];
    for &(t, c, k, want) in cases {
        let got = cursor_after(t, c, k);
        if got != want {
            bad.push(format!("{t:?}@{c} {k}: got {got}, want {want}"));
        }
    }
    assert!(bad.is_empty(), "{bad:#?}");
}

#[test]
fn dr01_guard_u_cursor_cases_that_already_agree() {
    assert_eq!(cursor_after("abc\n", 0, "i<Right>x<Esc>u"), 1);
    assert_eq!(cursor_after("abc\n", 1, "Ax<Esc>u"), 2);
    assert_eq!(cursor_after("  abc\n", 3, "Ix<Esc>u"), 2);
}

#[test]
fn dr02_motions_on_the_line_after_a_final_newline_stay_on_it() {
    for k in ["0", "^", "$"] {
        let (text, cur, _, msgs) = run("", 0, &format!("ifoo<CR><Esc>{k}"));
        assert_eq!(text, "foo\n");
        assert_eq!((cur, msgs.len()), (4, 0), "{k} from the empty last line");
    }
    let (_, cur, _, msgs) = run("", 0, "ifoo<CR><Esc>}");
    assert_eq!((cur, msgs.len()), (4, 1), "}} from the empty last line: {msgs:?}");
    let (text, cur, _, _) = run("", 0, "ifoo<CR><Esc>ibar<Esc>u");
    assert_eq!((text.as_str(), cur), ("foo\n", 4), "u back to the empty last line");
}

#[test]
fn dr04_insert_entered_for_another_window_puts_that_buffer_in_insert_state() {
    let mut ctx = Ctx::new(vec![Box::new(Vim::default())]).unwrap();
    let a = ctx.create_text("a", "aaa\n");
    let wa = ctx.open_window(a, 80, 24).unwrap();
    let b = ctx.create_text("b", "bbb\n");
    let wb = ctx.open_window(b, 80, 24).unwrap();
    ctx.focus(wa);
    let args = die_core::Args {
        window: Some(wb),
        buffer: Some(b),
        ..Default::default()
    };
    ctx.run("vim-insert", &args).unwrap();
    assert_eq!(ctx.state(), State::Normal, "A, focused, wasn't asked to insert");
    ctx.focus(wb);
    assert_eq!(ctx.state(), State::Insert, "B is the buffer in insert state");
}

#[test]
fn dr05_a_motion_that_falls_short_keeps_the_goal_column() {
    let (_, cur, _, msgs) = run("abcdefgh\nabc\nabcdefgh\n", 6, "j5hj");
    assert_eq!(msgs.len(), 1, "{msgs:?}");
    assert_eq!(cur, 19);
}

#[test]
fn dr06_insert_state_down_keeps_the_goal_column() {
    let (text, cur, _, _) = run("abcdef\nab\nabcdef\n", 4, "i<Down><Down>X<Esc>");
    assert_eq!(text, "abcdef\nab\nabcdXef\n");
    assert_eq!(cur, 14);
}

#[test]
fn dr06_extra() {
    let (text, cur, _, msgs) = run("abcdef\nab\nabcdef\n", 14, "i<Up><Up>X<Esc>");
    let (text2, cur2, _, _) = run("abcdef\nab\nabcdef\n", 4, "i<Down>X<Esc>");
    // goal from normal state j must not carry: j to col 1 of "ab"? start col 4 on line 0,
    // j -> "ab" col 1 (goal 4), i, <Down> -> Emacs recomputes goal from col 1
    let (text3, cur3, _, _) = run("abcdef\nab\nabcdef\n", 4, "ji<Down>X<Esc>");
    panic!("up-up: {text:?} {cur} {msgs:?}; down: {text2:?} {cur2}; j-i-down: {text3:?} {cur3}");
}

#[test]
fn dr07_insert_down_and_ctrl_a_follow_screen_rows_on_a_wrapped_line() {
    let long: Vec<String> = (0..24).map(|i| format!("word{i:02}")).collect();
    let text = format!("{}\nnext\n", long.join(" "));
    let ctrl_a = cursor_after(&text, 0, "$<C-a>");
    let down = run(&text, 0, "i<Down>X<Esc>").0.find('X');
    assert_eq!(
        (ctrl_a, down),
        (154, Some(77)),
        "(C-a's cursor, where insert-state <Down> typed X)"
    );
}

#[test]
fn dr09_a_large_count_insert_finishes_quickly() {
    let (tx, rx) = std::sync::mpsc::channel();
    std::thread::spawn(move || {
        let r = run("\n", 0, "50000ix<Esc>");
        let _ = tx.send(r.0.len());
    });
    match rx.recv_timeout(std::time::Duration::from_secs(2)) {
        Ok(len) => assert_eq!(len, 50001),
        Err(_) => panic!("50000ix<Esc> took more than 2 s"),
    }
}

#[test]
fn dr09_timing_big() {
    let mut out = vec![];
    for n in [100_000usize, 1_000_000, 4_000_000] {
        let t0 = std::time::Instant::now();
        let r = run("\n", 0, &format!("{n}ix<Esc>"));
        let t1 = t0.elapsed();
        // and one motion afterwards on that line
        let t2 = std::time::Instant::now();
        let r2 = run(&r.0, 0, "$");
        out.push(format!(
            "{n}: insert {t1:?} len {} msgs {:?}; then $ {:?} -> {}",
            r.0.len(),
            r.3,
            t2.elapsed(),
            r2.1
        ));
    }
    // a session that isn't text-only stays on the per-key path
    for n in [2000usize, 4000] {
        let t0 = std::time::Instant::now();
        let r = run("\n", 0, &format!("{n}ixy<Left><Esc>"));
        out.push(format!("{n}ixy<Left>: {:?} len {}", t0.elapsed(), r.0.len()));
    }
    panic!("{out:#?}");
}

#[test]
fn dr11_dollar() {
    let t = "cafe\u{301}\n";
    let (_, cur, _, _) = run(t, 0, "$");
    let (_, cur2, _, _) = run("a cafe\u{301}\n", 0, "we");
    let (_, cur3, _, _) = run("cafe\u{301} bar\n", 0, "e");
    panic!("$ -> {cur}; we -> {cur2}; e mid -> {cur3}");
}

#[test]
fn dr11_e_twice() {
    let t = "cafe\u{301}\nbar\n";
    let (_, cur, _, msgs) = run(t, 0, "ee");
    let (_, cur2, _, msgs2) = run("cafe\u{301}\n", 0, "ej");
    let (_, cur3, _, msgs3) = run("cafe\u{301} bar\n", 0, "ee");
    let e_eol = run("cafe\u{301}\n", 0, "e").1;
    let e_mid = run("cafe\u{301} bar\n", 0, "e").1;
    let dollar = run("cafe\u{301}\n", 0, "$").1;
    panic!(
        "ee -> {cur} {msgs:?}; mid ee -> {cur3} {msgs3:?}; ej {cur2} {msgs2:?}; e eol {e_eol}; e mid {e_mid}; $ {dollar}"
    );
}
