//! Deep-read review: die-vim cases where the owner's Evil (run through
//! tools/evil-oracle on 2026-11-03, results quoted per test) and die-vim differ,
//! plus invariants die-vim states and breaks. Every test here fails on main
//! except `u_cursor_cases_that_already_agree`, a guard for the `u` fix.

use die_core::{Ctx, State};
use die_vim::Vim;

/// Run `keys` on `text` with the cursor at byte `cursor`; return the text, the
/// cursor, the state and the messages.
fn run(text: &str, cursor: usize, keys: &str) -> (String, usize, State, Vec<String>) {
    let mut ctx = Ctx::new(vec![Box::new(Vim::default())]).expect("die-vim registers");
    let b = ctx.create_text("t", text);
    let w = ctx.open_window(b, 80, 24).expect("a window");
    ctx.focus(w);
    assert_eq!(ctx.text_mut(b).unwrap().set_cursor(w, cursor), Some(cursor));
    ctx.type_keys(keys);
    let t = ctx.text(b).unwrap();
    (
        t.rope().to_string(),
        t.cursor(w).unwrap(),
        ctx.state(),
        ctx.messages.clone(),
    )
}

fn cursor_after(text: &str, cursor: usize, keys: &str) -> usize {
    run(text, cursor, keys).1
}

// ---------------------------------------------------------------- u's cursor

/// Evil: `u` goes back to where the command that made the step's first change
/// started (Emacs records point before that command). die-vim puts the cursor
/// at the start of the undone range instead.
/// Evil oracle: "abc\n" @0 `ox<Esc>u` -> 0; "abc\ndef\n" @1 `ox<Esc>u` -> 1;
/// "abc\ndef\n" @5 `Ox<Esc>u` -> 5; "abc\n" @1 `3ox<Esc>u` -> 1;
/// "ab\ncd\n" @0 `jAx<Up>y<Esc>u` -> 4.
#[test]
fn u_returns_to_where_the_steps_first_change_was_made() {
    assert_eq!(
        cursor_after("abc\n", 0, "ox<Esc>u"),
        0,
        "ox<Esc>u from col 0"
    );
    assert_eq!(
        cursor_after("abc\ndef\n", 1, "ox<Esc>u"),
        1,
        "ox<Esc>u from col 1"
    );
    assert_eq!(
        cursor_after("abc\ndef\n", 5, "Ox<Esc>u"),
        5,
        "Ox<Esc>u from col 1"
    );
    assert_eq!(cursor_after("abc\n", 1, "3ox<Esc>u"), 1, "3ox<Esc>u");
    assert_eq!(
        cursor_after("ab\ncd\n", 0, "jAx<Up>y<Esc>u"),
        4,
        "a session's later change to the left"
    );
}

/// Guard for the fix: these already agree with Evil and must keep agreeing.
/// Evil oracle: "abc\n" @0 `i<Right>x<Esc>u` -> 1; "abc\n" @1 `Ax<Esc>u` -> 2;
/// "  abc\n" @3 `Ix<Esc>u` -> 2.
#[test]
fn u_cursor_cases_that_already_agree() {
    assert_eq!(cursor_after("abc\n", 0, "i<Right>x<Esc>u"), 1);
    assert_eq!(cursor_after("abc\n", 1, "Ax<Esc>u"), 2);
    assert_eq!(cursor_after("  abc\n", 3, "Ix<Esc>u"), 2);
}

// ---------------------------------------------------------------- the line after a final newline

/// V1-1 leaves the cursor on the empty line after a final newline when an
/// insert session ends there. From there `0`, `^` and `$` jump to the end of
/// the line above, `}` moves up without an error, and `u` leaves the line.
/// Evil oracle, "" @0: `ifoo<CR><Esc>0`, `^`, `$` -> 4 (line 1, col 0);
/// `ifoo<CR><Esc>}` -> 4 with "End of buffer"; `ifoo<CR><Esc>ibar<Esc>u` -> 4.
#[test]
fn motions_on_the_line_after_a_final_newline_stay_on_it() {
    for k in ["0", "^", "$"] {
        let (text, cur, _, msgs) = run("", 0, &format!("ifoo<CR><Esc>{k}"));
        assert_eq!(text, "foo\n");
        assert_eq!((cur, msgs.len()), (4, 0), "{k} from the empty last line");
    }
    let (_, cur, _, msgs) = run("", 0, "ifoo<CR><Esc>}");
    assert_eq!(
        (cur, msgs.len()),
        (4, 1),
        "}} from the empty last line: {msgs:?}"
    );
    let (text, cur, _, _) = run("", 0, "ifoo<CR><Esc>ibar<Esc>u");
    assert_eq!(
        (text.as_str(), cur),
        ("foo\n", 4),
        "u back to the empty last line"
    );
}

// ---------------------------------------------------------------- goal column

/// A motion that moves fewer times than its count signals an error, so Emacs
/// doesn't update `last-command` and the next `j` keeps the goal column, as
/// it does after a motion that fails outright (oracle case
/// motions/failed-l-keeps-goal-column). die-vim drops the goal.
/// Evil oracle: "abcdefgh\nabc\nabcdefgh\n" @6 `j5hj` -> 19 (line 2, col 6),
/// one error "Beginning of line".
#[test]
fn a_motion_that_falls_short_keeps_the_goal_column() {
    let (_, cur, _, msgs) = run("abcdefgh\nabc\nabcdefgh\n", 6, "j5hj");
    assert_eq!(msgs.len(), 1, "{msgs:?}");
    assert_eq!(cur, 19);
}

/// Insert state's <Down> keeps a goal column across a short line, as Emacs's
/// next-line does. die-vim takes the current column each time.
/// Evil oracle (tty run): "abcdef\nab\nabcdef\n" @4 `i<Down><Down>X<Esc>` ->
/// "abcdef\nab\nabcdXef\n", cursor 14.
#[test]
fn insert_state_down_keeps_the_goal_column() {
    let (text, cur, _, _) = run("abcdef\nab\nabcdef\n", 4, "i<Down><Down>X<Esc>");
    assert_eq!(text, "abcdef\nab\nabcdXef\n");
    assert_eq!(cur, 14);
}

/// Your Emacs has global-visual-line-mode and line-move-visual: in insert state
/// <Down> moves a screen row, and C-a (remapped to beginning-of-visual-line)
/// goes to the start of the screen row. die-vim moves by logical line for both.
/// Evil oracle (tty, 80 columns; this window is 80 columns and wraps the line
/// at the same places, 77 and 154): `$<C-a>` -> 154; `i<Down>X<Esc>` puts X at 77.
#[test]
fn insert_down_and_ctrl_a_follow_screen_rows_on_a_wrapped_line() {
    let long: Vec<String> = (0..24).map(|i| format!("word{i:02}")).collect();
    let text = format!("{}\nnext\n", long.join(" "));
    let ctrl_a = cursor_after(&text, 0, "$<C-a>");
    let down = run(&text, 0, "i<Down>X<Esc>").0.find('X');
    assert_eq!(
        (ctrl_a, down),
        (154, Some(77)),
        "(C-a's cursor, where insert-state <Down> typed X)"
    );
}

// ---------------------------------------------------------------- counts

/// An undefined key ends the command, and with it the count (Emacs's
/// `undefined` resets prefix-arg). In DIE the count survives to the next key.
/// Evil oracle: "abcdef\n" @0 `3gzl` -> 1 ("g z is undefined" is a message,
/// not an error).
#[test]
fn an_undefined_key_drops_the_count() {
    let (_, cur, _, msgs) = run("abcdef\n", 0, "3gzl");
    assert_eq!(cur, 1, "l after an undefined key moves once: {msgs:?}");
}

/// `C-a` is Emacs's move-beginning-of-line under your Evil: with a count it
/// moves count-1 lines down first. die-vim ignores the count.
/// Evil oracle: "abc\ndef\nghi\njkl\n" @2 `3<C-a>` -> 8 (line 2, col 0).
#[test]
fn count_ctrl_a_moves_down_count_minus_one_lines() {
    assert_eq!(cursor_after("abc\ndef\nghi\njkl\n", 2, "3<C-a>"), 8);
}

/// A huge count replays the session key by key, and each replayed key lays
/// out the whole (growing) line again: 8000ix<Esc> takes 0.73 s in release and
/// 1.0 s in debug, quadrupling as the count doubles. Vim inserts the text at
/// once. Nothing can interrupt it, so a mistyped count hangs the editor.
#[test]
fn a_large_count_insert_finishes_quickly() {
    let (tx, rx) = std::sync::mpsc::channel();
    std::thread::spawn(move || {
        let r = run("\n", 0, "50000ix<Esc>");
        let _ = tx.send(r.0.len());
    });
    match rx.recv_timeout(std::time::Duration::from_secs(2)) {
        Ok(len) => assert_eq!(len, 50001),
        Err(_) => panic!("50000ix<Esc> took more than 2 s"),
    }
}

// ---------------------------------------------------------------- words and scripts

/// Emacs ends a word where the script changes from Latin to Cyrillic (not
/// Latin to Greek). die-vim's scripts are Han, Kana, Hangul and Other, so
/// "abcабв" is one word.
/// Evil oracle: "abcабв def\n" @0 `w` -> 3, `e` -> 2; @7 `b` -> 3.
#[test]
fn a_word_ends_where_latin_meets_cyrillic() {
    let t = "abc\u{430}\u{431}\u{432} def\n";
    assert_eq!(cursor_after(t, 0, "w"), 3, "w");
    assert_eq!(cursor_after(t, 0, "e"), 2, "e");
    assert_eq!(cursor_after(t, 7, "b"), 3, "b");
}

// ---------------------------------------------------------------- the normal-state invariant

/// grammar.rs: "In normal state the cursor rests on a character, never on a
/// line's newline unless the line is empty". `e` onto a word that ends in a
/// combining mark at the end of a line lands on the newline: `normal_pos`
/// snaps the mark forward, past the grapheme, onto the line's end. (Your
/// Evil does the same, an Emacs redisplay artifact: oracle "café\n" `e` -> 6.)
#[test]
fn e_never_rests_on_a_non_empty_lines_newline() {
    let t = "cafe\u{301}\n";
    let (_, cur, state, _) = run(t, 0, "e");
    assert_eq!(state, State::Normal);
    assert_ne!(cur, 6, "the cursor is on the newline of \"café\"");
}

/// Emacs draws a control character as `^M`, two columns; die-core's layout
/// draws `?`, one column (layout.rs `put`). Goal columns after one are off by one.
/// Evil oracle: "a\rbc\nxyzwv\n" @3 `j` -> 9 (col 4); same with "\x01".
#[test]
fn a_control_character_is_two_columns_wide() {
    assert_eq!(cursor_after("a\rbc\nxyzwv\n", 3, "j"), 9, "after \\r");
    assert_eq!(cursor_after("a\x01bc\nxyzwv\n", 3, "j"), 9, "after \\x01");
}

// ---------------------------------------------------------------- state and the buffer acted on

/// die-vim's commands act on `Args::window`/`Args::buffer` (grammar.rs
/// `buffer_window`), but `Ctx::set_state` always sets the focused buffer's
/// state. Running `i` for a window that isn't focused (an extension, a macro
/// runner, later `:windo`) opens the session and its undo group on that
/// buffer, and puts the focused buffer into insert state instead. The group on
/// B then never closes: B's later edits all join one undo step.
#[test]
fn insert_entered_for_another_window_puts_that_buffer_in_insert_state() {
    let mut ctx = Ctx::new(vec![Box::new(Vim::default())]).unwrap();
    let a = ctx.create_text("a", "aaa\n");
    let wa = ctx.open_window(a, 80, 24).unwrap();
    let b = ctx.create_text("b", "bbb\n");
    let wb = ctx.open_window(b, 80, 24).unwrap();
    ctx.focus(wa);
    let args = die_core::Args {
        window: Some(wb),
        buffer: Some(b),
        ..Default::default()
    };
    ctx.run("vim-insert", &args).unwrap();
    assert_eq!(
        ctx.state(),
        State::Normal,
        "A, focused, wasn't asked to insert"
    );
    ctx.focus(wb);
    assert_eq!(
        ctx.state(),
        State::Insert,
        "B is the buffer in insert state"
    );
}

// ---------------------------------------------------------------- partial key sequences

/// Escape cancels a pending `f`, `F`, `t` or `T` in Vim and in Evil (which
/// quits the read silently; the batch oracle can't record it because the quit
/// ends the run). die-vim reports "f <Esc> is undefined", and the count stays
/// for the next command.
#[test]
fn escape_cancels_a_pending_f_quietly() {
    let (_, cur, _, msgs) = run("abcdef\n", 0, "3f<Esc>l");
    assert!(msgs.is_empty(), "{msgs:?}");
    assert_eq!(cur, 1);
}
