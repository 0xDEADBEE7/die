//! Deep-read review: undo attribution and grouping cases that fail on main.

use die_core::*;
use std::cell::RefCell;
use std::rc::Rc;

fn setup(text: &str) -> (Ctx, BufferId, WindowId) {
    let mut ctx = Ctx::new(vec![]).unwrap();
    let b = ctx.create_text("t", text);
    let w = ctx.open_window(b, 40, 10).unwrap();
    (ctx, b, w)
}

/// Run `f` as one top-level unit (a queued callback): one undo step by default (A11).
fn step<R: 'static>(ctx: &mut Ctx, b: BufferId, f: impl FnOnce(&mut TextBuf) -> R + 'static) -> R {
    let out = Rc::new(RefCell::new(None));
    let o = out.clone();
    ctx.defer(move |ctx| {
        *o.borrow_mut() = Some(f(ctx.text_mut(b)?));
        Ok(())
    });
    ctx.run_until_idle();
    out.borrow_mut().take().expect("the step ran")
}

fn insert(ctx: &mut Ctx, b: BufferId, at: usize, s: &'static str) {
    step(ctx, b, move |t| {
        t.tx(Origin::User).insert(at, s).commit().unwrap()
    });
}

fn text(ctx: &Ctx, b: BufferId) -> String {
    ctx.text(b).unwrap().rope().to_string()
}

/// `text_mut` stamps the buffer with `cmd_seq`, which only `step` and
/// `dispatch` advance. An edit made by host code between steps (the harness,
/// a file revert, an embedder) gets the last command's number and joins its
/// undo step, so one `u` takes back both.
#[test]
fn an_edit_made_outside_any_command_is_its_own_undo_step() {
    let (mut ctx, b, _) = setup("");
    insert(&mut ctx, b, 0, "a"); // a command
    ctx.text_mut(b)
        .unwrap()
        .tx(Origin::User)
        .insert(1, "b")
        .commit()
        .unwrap(); // host code, between commands
    assert_eq!(text(&ctx, b), "ab");
    step(&mut ctx, b, |t| t.undo()).unwrap();
    assert_eq!(text(&ctx, b), "a", "u took back the command's edit too");
}

/// An explicit group spans commands until `end_undo_group` (contract §9). An
/// undo inside it (Vim's `<C-o>u`, or any extension calling `undo`) resets the
/// group depth to 0 (`History::commit`), so every later edit of the session
/// is a step of its own and the session's `end_undo_group` does nothing.
#[test]
fn an_undo_inside_an_open_group_leaves_the_group_open() {
    let (mut ctx, b, _) = setup("");
    step(&mut ctx, b, |t| t.begin_undo_group());
    insert(&mut ctx, b, 0, "x");
    step(&mut ctx, b, |t| t.undo()).unwrap();
    assert_eq!(text(&ctx, b), "");
    insert(&mut ctx, b, 0, "y");
    insert(&mut ctx, b, 1, "z");
    step(&mut ctx, b, |t| t.end_undo_group());
    step(&mut ctx, b, |t| t.undo()).unwrap();
    assert_eq!(
        text(&ctx, b),
        "",
        "the rest of the group (yz) should be one step"
    );
}

/// Core restores the selection captured at `begin_undo_group`, not the one
/// before the group's first edit. die-vim's `i<Right>x<Esc>` begins the group
/// at 0 and types at 1; your Evil's `u` returns to 1 (oracle: "abc\n" @0
/// `i<Right>x<Esc>u` -> 1), as it returns to the point before the command that
/// made the first change in every case checked. die-vim can only use core's
/// restored selection for `u` if core captures it there.
#[test]
fn undo_restores_the_selection_from_before_the_groups_first_edit() {
    let (mut ctx, b, w) = setup("abc\n");
    step(&mut ctx, b, |t| t.begin_undo_group()); // `i`, cursor at 0
    step(&mut ctx, b, move |t| t.set_cursor(w, 1)); // <Right>
    insert(&mut ctx, b, 1, "x"); // x
    step(&mut ctx, b, |t| t.end_undo_group()); // <Esc>
    step(&mut ctx, b, |t| t.undo()).unwrap();
    assert_eq!(ctx.text(b).unwrap().cursor(w), Some(1));
}

/// Contract §9: "If a silent edit changed the text an undo step needs, undo
/// refuses with an error and drops the older history." When the silent edit
/// lands inside a group that is still collecting (an extension rewriting what
/// you're typing), `History::record` drops the whole history at your next
/// keystroke (`None => self.undo.clear()`), silently. The undo that reaches the
/// dropped part then says `NothingToUndo`, so nothing ever says why it's gone.
#[test]
fn history_dropped_by_a_conflict_while_typing_is_reported_as_a_conflict() {
    let (mut ctx, b, _) = setup("");
    insert(&mut ctx, b, 0, "older ");
    step(&mut ctx, b, |t| t.begin_undo_group());
    insert(&mut ctx, b, 6, "ab");
    step(&mut ctx, b, |t| {
        t.tx(Origin::Extension)
            .silent()
            .replace(7..8, "B")
            .commit()
            .unwrap()
    });
    insert(&mut ctx, b, 8, "c");
    step(&mut ctx, b, |t| t.end_undo_group());
    assert_eq!(text(&ctx, b), "older aBc");
    step(&mut ctx, b, |t| t.undo()).unwrap();
    assert_eq!(text(&ctx, b), "older aB");
    assert_eq!(
        step(&mut ctx, b, |t| t.undo()),
        Err(Error::UndoConflict),
        "the older steps were dropped by a conflict"
    );
}
