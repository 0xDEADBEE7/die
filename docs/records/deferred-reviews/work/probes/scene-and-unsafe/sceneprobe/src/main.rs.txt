//! Probes die-core's Scene for the Scene review: sizes, per-frame cost at
//! 300x100, and a few behaviours the tests don't pin.
use die_core::*;
use std::ops::Range;
use std::time::Instant;

const FACES: [&str; 6] = ["p.kw", "p.str", "p.com", "p.ty", "p.fun", "p.num"];

/// Declares the probe's faces and the token-painting mode.
struct Ext;
impl Extension for Ext {
    fn name(&self) -> &'static str {
        "p"
    }
    fn register(&mut self, reg: &mut Registry) {
        for (i, f) in FACES.iter().enumerate() {
            let s = Style::new().fg(Rgb(10 * i as u8 + 40, 80, 120));
            let s = if i % 2 == 0 { s.bold() } else { s.italic() };
            reg.face(f, None, s);
        }
        reg.mode::<Tokens>("tokens", Tokens::default);
    }
}

/// A syntax highlighter stand-in: one face per word in the visible text.
#[derive(Default)]
struct Tokens {
    produced: usize,
}
impl Mode for Tokens {
    fn decorate(
        &mut self,
        t: &TextBuf,
        visible: Range<usize>,
        out: &mut Vec<(Range<usize>, &'static str)>,
    ) {
        let text = t.rope().byte_slice(visible.clone()).to_string();
        let mut start = None;
        let mut k = 0;
        for (i, ch) in text.char_indices().chain([(text.len(), ' ')]) {
            match (start, ch.is_alphanumeric() || ch == '_') {
                (None, true) => start = Some(i),
                (Some(s), false) => {
                    out.push((visible.start + s..visible.start + i, FACES[k % FACES.len()]));
                    k += 1;
                    start = None;
                }
                _ => {}
            }
        }
        self.produced = k;
    }
}

fn source(dense: bool) -> String {
    let src = include_str!("/Users/user/workspace/die/crates/die-core/src/scene.rs");
    if !dense {
        return src.to_string();
    }
    // Every line about 280 columns of tokens: the worst realistic case for 300 columns.
    src.lines()
        .filter(|l| !l.trim().is_empty())
        .map(|l| {
            let mut s = String::new();
            while s.len() < 280 {
                s.push_str(l.trim());
                s.push(' ');
            }
            s.truncate(280);
            s
        })
        .collect::<Vec<_>>()
        .join("\n")
}

fn time(label: &str, n: usize, mut f: impl FnMut() -> Scene) {
    let mut v = vec![];
    let mut keep = None;
    for _ in 0..n {
        let t = Instant::now();
        let s = f();
        v.push(t.elapsed().as_secs_f64() * 1e3);
        keep = Some(s);
    }
    v.sort_by(|a, b| a.partial_cmp(b).unwrap());
    let s = keep.unwrap();
    println!(
        "{label:58} p50 {:7.3} ms  p99 {:7.3} ms  ({}x{})",
        v[n / 2],
        v[n * 99 / 100],
        s.cols,
        s.rows
    );
}

fn main() {
    println!("size_of Cell {}  Glyph {}  Option<(WindowId, usize)> {}  Cursor {}",
        std::mem::size_of::<Cell>(), std::mem::size_of::<Glyph>(),
        std::mem::size_of::<Option<(WindowId, usize)>>(), std::mem::size_of::<Cursor>());
    let per = std::mem::size_of::<Cell>() + std::mem::size_of::<Option<(WindowId, usize)>>();
    println!("300x100 scene: {} cells, {} bytes in cells + positions ({:.2} MB)", 30000, 30000 * per, (30000 * per) as f64 / 1e6);

    for dense in [false, true] {
        let mut ctx = Ctx::new(vec![Box::new(Ext)]).unwrap();
        let b = ctx.create_text("scene.rs", &source(dense));
        let (cols, rows) = (300, 100);
        let w = ctx.open_window(b, cols, rows - 2).unwrap();
        ctx.focus(w);
        ctx.text_mut(b).unwrap().settings_mut().line_numbers = LineNumbers::Relative;
        ctx.text_mut(b).unwrap().set_cursor(w, 2000);
        let place = [Placement { window: w, col: 0, row: 0 }];
        let tag = if dense { "dense 280-col lines" } else { "scene.rs as is" };
        time(&format!("{tag}: plain"), 200, || ctx.compose(cols, rows, &place));
        ctx.enable_mode::<Tokens>(b).unwrap();
        time(&format!("{tag}: + token provider"), 50, || ctx.compose(cols, rows, &place));
        let n = ctx.with_mode(b, |m: &mut Tokens, _| m.produced).unwrap();
        println!("    provider produced {n} face ranges");
        ctx.set_search_highlight(Some(Pattern::new("e", false).unwrap()));
        time(&format!("{tag}: + tokens + search \"e\""), 20, || ctx.compose(cols, rows, &place));
        ctx.set_search_highlight(None);
        let len = ctx.text_mut(b).unwrap().rope().len_bytes();
        ctx.set_state(State::Visual);
        ctx.text_mut(b).unwrap().set_selection(w, Selection { anchor: 0, head: len.min(12000), shape: Shape::Block, goal: None });
        time(&format!("{tag}: + tokens + visual block"), 10, || ctx.compose(cols, rows, &place));
        ctx.set_state(State::Normal);
    }

    // Search alone (M0 has no providers), dense text.
    {
        let mut ctx = Ctx::new(vec![Box::new(Ext)]).unwrap();
        let b = ctx.create_text("d", &source(true));
        let w = ctx.open_window(b, 300, 98).unwrap();
        ctx.focus(w);
        let place = [Placement { window: w, col: 0, row: 0 }];
        ctx.set_search_highlight(Some(Pattern::new("e", false).unwrap()));
        time("dense: search \"e\" only (no provider)", 20, || ctx.compose(300, 100, &place));
        ctx.set_search_highlight(Some(Pattern::new("self", false).unwrap()));
        time("dense: search \"self\" only (no provider)", 20, || ctx.compose(300, 100, &place));
        let w2 = ctx.open_window(b, 80, 48).unwrap();
        ctx.focus(w2);
        let place = [Placement { window: w2, col: 0, row: 0 }];
        ctx.set_search_highlight(Some(Pattern::new("e", false).unwrap()));
        time("dense, 80x50 wrapped: search \"e\" only", 20, || ctx.compose(80, 50, &place));
        ctx.set_search_highlight(None);
        ctx.enable_mode::<Tokens>(b).unwrap();
        time("dense, 80x50 wrapped: token provider only", 20, || ctx.compose(80, 50, &place));
    }

    // The owner's layout: an 80-column text area, full height, dense wrapped text.
    {
        let mut ctx = Ctx::new(vec![Box::new(Ext)]).unwrap();
        let b = ctx.create_text("d", &source(true));
        let w = ctx.open_window(b, 84, 98).unwrap();
        ctx.focus(w);
        ctx.text_mut(b).unwrap().settings_mut().line_numbers = LineNumbers::Relative;
        let place = [Placement { window: w, col: 0, row: 0 }];
        time("80x98 text area, dense wrapped: plain", 50, || ctx.compose(84, 100, &place));
        ctx.set_search_highlight(Some(Pattern::new("e", false).unwrap()));
        time("80x98 text area, dense wrapped: search \"e\"", 50, || ctx.compose(84, 100, &place));
        ctx.set_search_highlight(None);
        ctx.enable_mode::<Tokens>(b).unwrap();
        time("80x98 text area, dense wrapped: token provider", 50, || ctx.compose(84, 100, &place));
    }

    // What R2's row diff costs: compare each row with the previous frame's, or hash it.
    {
        let mut ctx = Ctx::new(vec![Box::new(Ext)]).unwrap();
        let b = ctx.create_text("d", &source(true));
        let w = ctx.open_window(b, 300, 98).unwrap();
        ctx.focus(w);
        let place = [Placement { window: w, col: 0, row: 0 }];
        let a = ctx.compose(300, 100, &place);
        let c = ctx.compose(300, 100, &place);
        let t = Instant::now();
        let mut same = 0;
        for _ in 0..100 { same += (0..a.rows).filter(|&r| a.row(r) == c.row(r)).count(); }
        println!("compare 100 rows with the previous frame: {:.3} ms per frame ({} same)", t.elapsed().as_secs_f64() * 1e3 / 100.0, same / 100);
        use std::hash::{Hash, Hasher};
        let t = Instant::now();
        let mut x = 0u64;
        for _ in 0..100 { for r in 0..a.rows { let mut h = std::collections::hash_map::DefaultHasher::new(); a.row(r).hash(&mut h); x ^= h.finish(); } }
        println!("SipHash 100 rows: {:.3} ms per frame ({x:x})", t.elapsed().as_secs_f64() * 1e3 / 100.0);
        let t = Instant::now();
        for _ in 0..100 { let k = a.clone(); std::hint::black_box(k); }
        println!("clone a 300x100 Scene: {:.3} ms", t.elapsed().as_secs_f64() * 1e3 / 100.0);
    }

    // Behaviours.
    let mut ctx = Ctx::new(vec![Box::new(Ext)]).unwrap();
    let b = ctx.create_text("t", "ab\n\ncd\n");
    let w = ctx.open_window(b, 10, 4).unwrap();
    ctx.focus(w);
    let s = ctx.compose(10, 6, &[Placement { window: w, col: 0, row: 0 }]);
    let row = |r: usize| (0..10).map(|c| s.position(r, c).map(|p| p.1)).collect::<Vec<_>>();
    println!("positions row 0 ('ab'): {:?}", row(0));
    println!("positions row 1 (empty line): {:?}", row(1));
    println!("positions row 3 (after the last newline): {:?}", row(3));

    // Linewise selection: is the newline's cell drawn as selected?
    ctx.set_state(State::Visual);
    ctx.text_mut(b).unwrap().set_selection(w, Selection { anchor: 0, head: 0, shape: Shape::Line, goal: None });
    let s = ctx.compose(10, 6, &[Placement { window: w, col: 0, row: 0 }]);
    let sel = ctx.face_style("selection").unwrap().bg.unwrap();
    println!("linewise selection on 'ab', cells selected: {:?}", (0..10).map(|c| s.row(0)[c].bg == sel).collect::<Vec<_>>());
    ctx.text_mut(b).unwrap().set_selection(w, Selection { anchor: 0, head: 4, shape: Shape::Line, goal: None });
    let s = ctx.compose(10, 6, &[Placement { window: w, col: 0, row: 0 }]);
    for r in 0..3 {
        println!("linewise selection over 3 lines, row {r} {:?}: {}", s.row_text(r), (0..10).map(|c| if s.row(r)[c].bg == sel { '#' } else { '.' }).collect::<String>());
    }
    ctx.set_state(State::Normal);

    // Hide replacement and virtual text: can a face reach them?
    let b2 = ctx.create_text("h", "fn x() { body }\n");
    let w2 = ctx.open_window(b2, 30, 1).unwrap();
    ctx.focus(w2);
    {
        let t = ctx.text_mut(b2).unwrap();
        t.decorate(8..14, Deco::Hide { replacement: Some("...".into()), reveal: Reveal::Never }).unwrap();
        t.decorate(0..15, Deco::Face("p.kw")).unwrap();
        t.decorate(3..3, Deco::Virtual("VT".into())).unwrap();
        t.set_cursor(w2, 0);
    }
    let s = ctx.compose(30, 3, &[Placement { window: w2, col: 0, row: 0 }]);
    let kw = ctx.face_style("p.kw").unwrap().fg.unwrap();
    println!("row: {:?}", s.row_text(0));
    println!("cells in p.kw's fg: {}", (0..30).map(|c| if s.row(0)[c].fg == kw { '#' } else { '.' }).collect::<String>());

    // Snapshot letters past 62 distinct styles.
    let mut ctx = Ctx::new(vec![]).unwrap();
    let text: String = (0..70).map(|_| "x").collect::<Vec<_>>().join("");
    let b3 = ctx.create_text("s", &format!("{text}\n"));
    let w3 = ctx.open_window(b3, 70, 1).unwrap();
    ctx.focus(w3);
    let mut s = ctx.compose(70, 3, &[Placement { window: w3, col: 0, row: 0 }]);
    for c in 0..70 {
        s.cells[c].fg = Rgb(c as u8, 0, 0);
    }
    let snap = s.snapshot();
    let style_row = snap.lines().find(|l| l.starts_with("|") && l.contains('?')).unwrap_or("");
    println!("70 distinct styles, style row: {style_row}");
    println!("legend lines starting with '?': {}", snap.lines().filter(|l| l.starts_with("? ")).count());
}
