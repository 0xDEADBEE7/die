use objc2_core_foundation::{CFArray, CFRetained, CFString, CFURL, CFURLPathStyle};
use objc2_core_text::{CTFont, CTFontDescriptor, CTFontManagerCreateFontDescriptorsFromURL};

fn descs(path: &str) -> CFRetained<CFArray> {
    let s = CFString::from_str(path);
    let url = CFURL::with_file_system_path(None, Some(&s), CFURLPathStyle::CFURLPOSIXPathStyle, false).unwrap();
    unsafe { CTFontManagerCreateFontDescriptorsFromURL(&url) }.unwrap()
}

/// What text.rs does today.
fn today(path: &str, px: f64) -> CFRetained<CTFont> {
    let descs = descs(path);
    assert!(descs.count() > 0);
    let desc = unsafe { &*(descs.value_at_index(0) as *const CTFontDescriptor) };
    unsafe { CTFont::with_font_descriptor(desc, px, std::ptr::null()) }
}

/// The proposed form: a typed array, and a checked, retained `get`.
fn proposed(path: &str, px: f64) -> Option<CFRetained<CTFont>> {
    // SAFETY: CTFontManagerCreateFontDescriptorsFromURL returns an array of
    // CTFontDescriptors.
    let descs: CFRetained<CFArray<CTFontDescriptor>> = unsafe { CFRetained::cast_unchecked(descs(path)) };
    let desc = descs.get(0)?;
    // SAFETY: a null matrix means the identity transform.
    Some(unsafe { CTFont::with_font_descriptor(&desc, px, std::ptr::null()) })
}

fn show(label: &str, f: &CTFont) {
    unsafe {
        println!("{label:28} ps={} ascent={:.3} descent={:.3}", f.post_script_name(), f.ascent(), f.descent());
    }
}

fn main() {
    let args: Vec<String> = std::env::args().skip(1).collect();
    for p in &args {
        show(&format!("today {}", p.rsplit('/').next().unwrap()), &today(p, 32.0));
        show(&format!("proposed {}", p.rsplit('/').next().unwrap()), &proposed(p, 32.0).unwrap());
    }
}

/// The proposed `Face::glyph`: pointers to the whole buffers, not to their first elements.
pub fn glyph(font: &CTFont, ch: char) -> Option<u16> {
    use std::ptr::NonNull;
    let mut chars = [0u16; 2];
    let chars = ch.encode_utf16(&mut chars);
    let mut glyphs = [0u16; 2];
    // SAFETY: both pointers cover `chars.len()` (1 or 2) elements, as the call requires.
    let ok = unsafe {
        font.glyphs_for_characters(
            NonNull::from(&mut *chars).cast(),
            NonNull::from(&mut glyphs).cast(),
            chars.len() as isize,
        )
    };
    (ok && glyphs[0] != 0).then_some(glyphs[0])
}

#[test]
fn glyphs_bmp_and_astral() {
    let term = proposed("/Users/user/workspace/die/fonts/IoskeleyMonoTerm-Regular.ttf", 32.0).unwrap();
    assert!(glyph(&term, 'M').is_some());
    assert_eq!(glyph(&term, '\u{10FFFD}'), None);
    let emoji = proposed("/System/Library/Fonts/Apple Color Emoji.ttc", 32.0).unwrap();
    assert!(glyph(&emoji, '😀').is_some(), "a surrogate pair maps");
}
